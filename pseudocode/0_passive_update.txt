# dt = time since the previous step: DT, or longer when fast_forward jumps over a stationary regime.
# During a jump, births and self-destructions are forced by the jump (drawn from their exact
# waiting-time laws) instead of rolled here, and tech/population follow their exact growth law.
if rng.random() < CIV_BIRTH_RATE * dt:
    world.civs.append(new_civ(sample_lens_position(config, rng), config, rng))

for civ in world.civs:
    tech_growth(civ, dt)
    population_growth(civ, dt)   # capacity = POPULATION_PER_WORLD * civ.nb_of_world
    civ.broadcast_radius = compute_broadcast_radius(civ, now)
    civ.probe_speed = compute_probe_speed(civ.tech_level, civ.probe_speed)
    advance_probes(civ, dt)  # records when the probe sphere reaches the Sun
    # settlement probes hop to the nearest habitable world of the civ's type (mean distance
    # 0.554 * density^(-1/3)) at the probe speed; a settled world launches its own after
    # COLONIZATION_LAUNCH_PERIOD_YEARS (Carroll-Nellenback et al. 2019)
    hop = 0.554 * world_density(civ.biochem) ^ (-1/3)
    front_speed = civ.colonization_aggressiveness * hop / (hop / civ.probe_speed + COLONIZATION_LAUNCH_PERIOD_YEARS)
    advance[civ] = front_speed * dt

# colonization fronts only take empty worlds. R is the reach of the front in the free directions.
# Two competitors (same biochem) whose spheres meet stay in contact; each hides from the other the
# cap its sphere covers. The territory only grows in the free share of the sky (caps taken as
# independent). A civ inside the sphere of a competitor it never met (an enclave) cannot grow.
for civ in world.civs:
    new_R[civ] = civ.R + advance[civ]
for civ in world.civs with advance[civ] > 0:
    free[civ] = 1
    for each active competitor other with d = dist(civ, other):
        if other not in civ.contacts:
            if d <= other.R: free[civ] = 0 (enclave); stop
            if d <= civ.R or d > new_R[civ] + new_R[other]: skip
            civ.contacts += other; other.contacts += civ
        s = min(1, new_R[other] / d)
        free[civ] *= 1 - (1 - sqrt(1 - s^2)) / 2
for civ in world.civs with advance[civ] > 0 and free[civ] > 0:
    civ.nb_of_world += free[civ] * HABITABLE_WORLD_DENSITY * fraction_of_worlds(civ.biochem) * 4/3 pi (new_R[civ]^3 - civ.R^3)
    civ.R = new_R[civ]
    civ.record_snapshot(now)

# orphan probes keep spreading for PROBE_LIFETIME_PER_TECH * tech_at_death years
for civ in world.extinct_with_live_probes(now):
    advance_probes(civ, dt)

# dead civs stay visible from afar until their last light has crossed the region
index = SpatialIndex(world.visible(now).position)
for civ in world.active(now):
    reach = max(WIDEST_BROADCAST_EVER + civ.R + widest_visible_front, civ.probe_radius)
    for other in index.neighbors_within(civ.position, reach):
        if other is civ:
            continue
        d = dist(civ, other)
        view = public_view(other, at=now - d / C)  # None if not yet born or already dead at that time
        if view is None:
            continue
        gap = d - civ.R - view.R                                  # from sphere edge to sphere edge
        by_probe = public_view(civ, at=now - d / C).probe_radius >= d
        # humanity only sees through its probes here (its signal detection: final_silence_check)
        by_signal = not civ.is_human and view.leak_radius >= gap  # leak: continuous, certain
        if not civ.is_human and not (by_signal or by_probe) and view.beacon_radius >= gap:
            # beacon: threshold Exp(1) drawn once per pair, hazard accumulated while exposed
            exposure = civ.exposures.setdefault(other, [exponential(1), 0])
            exposure.hazard += -view.nb_of_world * log(1 - LISTEN_COVERAGE * BEACON_DUTY) / INTERACTION_PERIOD * dt
            by_signal = exposure.hazard >= exposure.threshold
        if not (by_signal or by_probe):
            continue
        if other not in civ.belief:
            civ.belief[other] = new_belief(view, trust=default_trust(civ, other))
        else:
            civ.belief[other].refresh_estimates(view)
        # technological explosion: what the target may have gained while its light travelled,
        # at the growth rate seen with it (exact solution of the tech growth law)
        civ.belief[other].tech_level_uncertainty = grow(view.tech_level, view.tech_rate, TECH_CAP, d / C) - view.tech_level
        civ.belief[other].last_seen_at = now
        civ.belief[other].channel = "signal" if by_signal else "probe"
        # witness of an ongoing war (victim hit during the step of the view): REPUTATION_TRUST_DROP per interaction period
        # (only if the victim identified the attacker)
        if view.attack_attributed and view.attacked_by in civ.belief and view.attacked_at > view.t - DT:
            civ.belief[view.attacked_by].trust = max(0, civ.belief[view.attacked_by].trust - REPUTATION_TRUST_DROP * DT / INTERACTION_PERIOD_YEARS)

for civ in world.active(now):
    if self_destruction(civ, rng) or civ.tech_level <= 0 or civ.population_size <= 0:
        civ.go_extinct(at=now)

roll_artefacts(world, until=now)
check_silence(world, now)
