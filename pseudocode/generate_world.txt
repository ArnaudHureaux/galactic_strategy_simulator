def generate_world(config, seed):
    # the whole galaxy is drawn with the same random numbers, only the region is kept:
    # regions of different sizes are nested (same civs, initial and born, inside the smaller one)
    civs = seed_life_in_bubbles(stream(seed, "world"), seed)
    civs += draw_births(stream(seed, "birth"), seed)   # take part once their birth_year comes
    humanity = make_humanity(config, stream(seed, "humanity"))
    civs.append(humanity)
    return World(civs=civs, humanity=humanity, now=config.present_date_years - config.start_lookback_years)
