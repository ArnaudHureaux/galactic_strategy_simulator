def tech_growth(civ, dt):
    if civ.is_human:
        # observed trajectory, exact solution of the growth law (no Euler step: its rate can reach 5e-4 / year)
        civ.tech_level = grow(INITIAL_TECH_LEVEL, civ.tech_progress_rate, TECH_CAP, now - civ.birth_year)
    elif TECH_GROWTH_MODEL == "logistic":
        tech_growth_logistic(civ, dt)
    else:
        tech_growth_exponential(civ, dt)

def tech_growth_logistic(civ, dt):
    r = civ.natural_tech_progress * (1 - civ.colonization_aggressiveness) * civ.population_size ** TECH_POPULATION_EXPONENT
    civ.tech_level += r * civ.tech_level * (1 - civ.tech_level / TECH_CAP) * dt

def tech_growth_exponential(civ, dt):
    r = civ.natural_tech_progress * (1 - civ.colonization_aggressiveness) * civ.population_size ** TECH_POPULATION_EXPONENT
    civ.tech_level += r * civ.tech_level * dt
