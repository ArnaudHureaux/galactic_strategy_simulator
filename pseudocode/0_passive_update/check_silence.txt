def check_silence(world, now):
    human = world.humanity
    contacted = len(human.belief) > 0 or human.attacked_by is not None
    # Hart's argument: a colonization front holding the Sun since the Solar System exists would be seen
    colonized = now >= PRESENT_DATE - ARTEFACT_WINDOW and any(
        holds(civ, human.position) for civ in world.active(now) - {human})
    # holds: dist(civ, point) <= civ.R and point outside the shadow cone, asin(other.R / d) wide,
    # of every active competitor other in contact with civ
    if now < PRESENT_DATE and (contacted or colonized):
        world.rejected = True


# contact (b): one discovery draw per civ whose probes sat near the Sun during the artefact window
def roll_artefacts(world, until):
    for civ in world.civs - {world.humanity}:
        presence = [civ.probe_arrived_at_human, civ.probes_end_at]  # end = inf while creator alive
        if not civ.artefact_rolled and overlaps(presence, [PRESENT_DATE - ARTEFACT_WINDOW, until]):
            civ.artefact_rolled = True
            # advanced owners make discreet probes; humanity searches today, with its present tech
            stealth = ARTEFACT_STEALTH_PER_TECH * max(0, tech_at_first_presence(civ) - HUMANITY_PRESENT_TECH)
            if rng.random() < ARTEFACT_DETECTION_PROBABILITY * exp(-stealth):
                world.rejected = True


# run once at PRESENT_DATE, independent of dt
def final_silence_check(world):
    roll_artefacts(world, until=PRESENT_DATE)
    # humanity's surveys: light arriving while humanity listens, above the survey sensitivity,
    # caught with the coverage c(d) of the surveys (one draw per exposed civ, detection stream)
    for civ in world.civs - {world.humanity}:
        d = dist(civ, world.humanity) - world.humanity.R
        emitted = [PRESENT_DATE - LISTENING_YEARS - dist(civ, world.humanity) / C, PRESENT_DATE - dist(civ, world.humanity) / C]
        views = public_views(civ, during=emitted)
        leak_n = max(view.nb_of_world for view in views if view.leak_radius >= d - view.R)      # 0 if none
        beacon_n = max(view.nb_of_world for view in views if view.beacon_radius >= d - view.R)  # 0 if none
        c = seti_coverage(distance of the nearest exposed system)   # 1 within 5.1 pc, 0.037 within 50 pc, 4e-4 beyond
        miss = (1 - c) ^ leak_n * (1 - c * BEACON_DUTY) ^ beacon_n
        if random() >= miss:
            world.rejected = True
