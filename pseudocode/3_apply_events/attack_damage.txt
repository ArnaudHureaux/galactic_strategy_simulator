# A war destroys min(1, DAMAGE_COEF * power ratio) of the target per interaction period; the power
# ratio follows the energy law anchored on humanity (EIRP_HUMAN_W, EIRP_CAP_W): each tech level of
# advance multiplies it by the same factor. An attack event stands for one step of it
# (step_fraction: see share_tech_update.txt). 1 = destroyed at once.
def attack_damage(power, target_tech_level):
    k = log(EIRP_CAP_W / EIRP_HUMAN_W) / (TECH_CAP - HUMANITY_PRESENT_TECH)
    return step_fraction(min(1, DAMAGE_COEF * exp(k * (power - target_tech_level))))

# A more advanced attacker hides better: probability that the victim identifies it.
def attribution_probability(power, target_tech_level):
    return exp(-ATTACK_STEALTH_PER_TECH * max(0, power - target_tech_level))
