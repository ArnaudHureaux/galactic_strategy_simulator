# Shares and wars are continuous: their rates are per INTERACTION_PERIOD_YEARS and one event
# applies one step (DT) of them, so their yearly effect does not depend on DT.
def step_fraction(rate):
    return 1 - (1 - rate) ** (DT / INTERACTION_PERIOD_YEARS)


def share_tech_update(tech_level, shared_tech_level):
    gap = shared_tech_level - tech_level
    if gap > 0:
        updated = tech_level + step_fraction(SHARE_CATCHUP) * gap
    else:
        updated = tech_level + SHARE_BACKFLOW * (DT / INTERACTION_PERIOD_YEARS) * shared_tech_level * exp(gap / SHARE_DECAY)
    return min(TECH_CAP, updated)
