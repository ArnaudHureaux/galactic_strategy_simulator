# SHARE_* rates are per SHARE_PERIOD_YEARS; one share event applies one step (DT) of them,
# so the yearly effect of sharing does not depend on DT.
def share_fraction(rate):
    return 1 - (1 - rate) ** (DT / SHARE_PERIOD_YEARS)


def share_tech_update(tech_level, shared_tech_level):
    gap = shared_tech_level - tech_level
    if gap > 0:
        updated = tech_level + share_fraction(SHARE_CATCHUP) * gap
    else:
        updated = tech_level + SHARE_BACKFLOW * (DT / SHARE_PERIOD_YEARS) * shared_tech_level * exp(gap / SHARE_DECAY)
    return min(TECH_CAP, updated)
