def seed_life_in_bubbles(config, rng):
    civs = []
    for _ in range(sample_poisson(config.expected_bubbles, rng)):
        center = sample_lens_position(config, rng)
        for _ in range(sample_poisson(config.civs_per_bubble, rng)):
            offset = sample_ball(config.panspermia_range_pc, rng)
            civs.append(new_civ(add(center, offset), config, rng))
    return civs
