def seed_life_in_bubbles(rng, seed):
    # bubble density given per kpc^2 at the Sun's radius, over the whole proposal disc
    civs = []
    for i in range(poisson(BUBBLE_DENSITY_KPC2 * proposal_disc_area_kpc2, rng)):
        center, kept = sample_galactic_position(rng)
        if not kept or not in_region(center, margin=PANSPERMIA_RANGE_PC):
            continue
        bubble_rng = substream(seed, f"bubble/{i}")   # same civs whatever the region size
        for k in range(poisson(CIVS_PER_BUBBLE, bubble_rng)):
            offset = sample_ball(PANSPERMIA_RANGE_PC, bubble_rng)
            # born at start + U(0, BIRTH_SPREAD_FRACTION * START_LOOKBACK_YEARS)
            birth = START_DATE + uniform(0, BIRTH_SPREAD_FRACTION * START_LOOKBACK_YEARS, bubble_rng)
            civ = new_civ(f"c_{i}_{k}", add(center, offset), bubble_rng, birth)
            if in_region(civ.position):
                civs.append(civ)
    return civs

def draw_births(rng, seed):
    # every later birth, drawn up front: Poisson process in time over the whole annulus
    births = []
    t = START_DATE
    for j in count():
        t += exponential(1 / (CIV_BIRTH_RATE_KPC2 * proposal_disc_area_kpc2), rng)
        if t >= PRESENT:
            return births
        position, kept = sample_galactic_position(rng)
        if kept and in_region(position):
            births.append(new_civ(f"b_{j}", position, substream(seed, f"birth/{j}"), t))
