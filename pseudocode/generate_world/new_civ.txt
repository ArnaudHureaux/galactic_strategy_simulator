def new_civ(position, config, rng):
    return Civilisation(
        position=position,
        birth_year=PRESENT_DATE_YEARS - START_LOOKBACK_YEARS,   # last sterilizing GRB
        status="alive",
        biochem_type=sample_biochem(config, rng),
        tech_level=INITIAL_TECH_LEVEL,
        population_size=INITIAL_POPULATION,
        # each civ's own rates: mean * U(1 - SPREAD, 1 + SPREAD)
        natural_tech_progress=NATURAL_TECH_PROGRESS_MEAN * rng.uniform(1 - NATURAL_TECH_PROGRESS_SPREAD, 1 + NATURAL_TECH_PROGRESS_SPREAD),
        natural_population_growth=NATURAL_POPULATION_GROWTH_MEAN * rng.uniform(1 - NATURAL_POPULATION_GROWTH_SPREAD, 1 + NATURAL_POPULATION_GROWTH_SPREAD),
        strategy=sample_strategy(rng),
        belief={},
    )


# the galaxy is a mix of archetypes (strategy_presets.json), in proportions STRATEGY_MIX_*;
# "random" draws every trait uniformly in its range
def sample_strategy(rng):
    archetype = rng.weighted_choice(ARCHETYPES, weights=STRATEGY_MIX)
    strategy = {"archetype": archetype, "broadcast": archetype.get("broadcast", rng.random() < BROADCAST_PROBABILITY)}
    for trait, (low, high) in STRATEGY_RANGES.items():
        if trait in archetype:   # colonization_aggressiveness is never pinned
            noise = STRATEGY_NOISE * (high - low) * rng.uniform(-1, 1)
            strategy[trait] = clip(archetype[trait] + noise, low, high)
        else:
            strategy[trait] = rng.uniform(low, high)
    return strategy
