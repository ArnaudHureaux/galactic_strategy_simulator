events = []

# one attack event per step and per ongoing war; ATTACK_COST_TECH is per interaction period
for target in civ_to_attack:
    # the attacker spends this share of its power: ln(1 - cost) / k tech levels (see damaged_tech)
    my_civ.tech_level = damaged_tech(my_civ.tech_level, step_fraction(ATTACK_COST_TECH))
    d = dist(my_civ, target)
    events.append(Event(
        kind="attack",
        src=my_civ.id,
        dst=target.id,
        t=now + d / C,
        power=compute_attack_power(my_civ.tech_level),
    ))

for target in civ_to_share:
    d = dist(my_civ, target)
    events.append(Event(
        kind="share",
        src=my_civ.id,
        dst=target.id,
        t=now + d / C,
        shared_tech_level=my_civ.tech_level,
    ))

# no travel: applied once every civ has decided, at the end of the current step
for target in civ_to_share_implicitly:
    events.append(Event(kind="implicit_share", src=my_civ.id, dst=target.id, t=now))
